import java.util.Scanner;

//TIP To <b>Run</b> code, press <shortcut actionId="Run"/> or
// click the <icon src="AllIcons.Actions.Execute"/> icon in the gutter.
public class Main {
    public static void main(String[] args) {
        //TIP Press <shortcut actionId="ShowIntentionActions"/> with your caret at the highlighted text
        // to see how IntelliJ IDEA suggests fixing it.

        Scanner scan = new Scanner(System.in);

        System.out.println("name");
        String name = scan.nextLine().trim();

        System.out.println("nazwisko");
        String surname = scan.nextLine().trim();

        System.out.println("e-mail");
        String email = scan.nextLine().trim();

        System.out.println("hasło");
        String password = scan.nextLine().trim();

        if(name.isEmpty() || surname.isEmpty() || email.isEmpty() || password.isEmpty()){
            System.out.println("Usupełnij pole");
        }else if(email.contains("@")  || email.contains(".") ) {
            System.out.println("E-mail musi zawierać @ i .");
        }else if(password.length() < 8  || password.matches("[a-z]") || password.matches("[A-Z]") || password.matches(".*[a-zA-Z0-9].*")) {
            System.out.println("Hasło musi zawierać minimum 8 znaków, D/M literę i znak specjalny");
        }else{
            System.out.println("GG");
        }

    }
}
