wyjszcie = False

while not wyjszcie:
    try:
        print("1 - Wyswietl liste 2 - Posortuj liste 3 - usun ucznia 4 - zakoncz ")
        wybor = int(input("Twoj wybor: "))
    except ValueError:
        print("wpisz liczbe")
        continue
    if wybor == 1:
        if PLIK.exists():
            with PLIK.open("r", encoding="utf-8") as f:
                for linia in f:
                    print(linia.strip())
        else:
            print("Plik uczniowie.txt nie istnieje")

    elif wybor == 2:
        if PLIK.exists():
            with PLIK.open("r", encoding="utf-8") as f:
                posortowane_linie = sorted(f.readlines())
            with PLIK_NOWY.open("w", encoding="utf-8") as newf:
                newf.writelines(posortowane_linie)
            print("Zapisano posortowane linie do uczniowie.txt")
        else:
            print("Plik uczniowie.txt nie istnieje")

    elif wybor == 3:
        wybor_dezyngrowanego = input("Wpisz ucznia dla usuniecia:")
        if PLIK.exists():
            with PLIK.open( "r", encoding="utf-8") as f:
                lines = f.readlines()

            with PLIK_NOWY.open( "w", encoding="utf-8") as f:
                for index, line in enumerate(lines, start = 1):
                    if index != wybor_dezyngrowanego:
                        f.write(line)

    elif wybor == 4:
        wyjszcie = True
        print("wyjscie")

    else:
        print("error")
